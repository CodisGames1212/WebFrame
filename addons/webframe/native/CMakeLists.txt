cmake_minimum_required(VERSION 3.24)
project(webframe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
set(GODOTCPP_PRECISION single CACHE STRING "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "" FORCE)
FetchContent_Declare(godot_cpp
    GIT_REPOSITORY https://github.com/godotengine/godot-cpp.git
    GIT_TAG godot-4.4-stable GIT_SHALLOW TRUE
    PATCH_COMMAND ${CMAKE_COMMAND} "-DGODOT_CPP_SOURCE=<SOURCE_DIR>"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/patch_godot_cpp.cmake")
FetchContent_MakeAvailable(godot_cpp)
add_library(webframe SHARED src/backend.cpp)
target_link_libraries(webframe PRIVATE godot-cpp::template_release)
if(WIN32)
    # Build-time SDK only; the static loader removes a separate loader DLL.
    FetchContent_Declare(webview2 URL https://www.nuget.org/api/v2/package/Microsoft.Web.WebView2/1.0.2903.40 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(webview2)
    target_sources(webframe PRIVATE src/windows.cpp)
    target_include_directories(webframe PRIVATE "${webview2_SOURCE_DIR}/build/native/include")
    if(CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64" OR CMAKE_SYSTEM_PROCESSOR MATCHES "ARM64|arm64|aarch64")
        set(WEBFRAME_ARCH arm64)
        set(WEBVIEW_ARCH arm64)
    elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
        set(WEBFRAME_ARCH x86_64)
        set(WEBVIEW_ARCH x64)
    else()
        set(WEBFRAME_ARCH x86_32)
        set(WEBVIEW_ARCH x86)
    endif()
    target_link_libraries(webframe PRIVATE "${webview2_SOURCE_DIR}/build/native/${WEBVIEW_ARCH}/WebView2LoaderStatic.lib" ole32 oleaut32 user32 shlwapi version)
    set(WEBFRAME_PLATFORM windows)
elseif(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    enable_language(OBJCXX)
    target_sources(webframe PRIVATE src/ios.mm)
    target_compile_options(webframe PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
    target_link_libraries(webframe PRIVATE "-framework UIKit" "-framework WebKit")
    set(WEBFRAME_PLATFORM ios)
    set(WEBFRAME_ARCH arm64)
    set_target_properties(webframe PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_RPATH TRUE)
elseif(APPLE)
    enable_language(OBJCXX)
    target_sources(webframe PRIVATE src/macos.mm)
    target_compile_options(webframe PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
    target_link_libraries(webframe PRIVATE "-framework Cocoa" "-framework WebKit")
    set(WEBFRAME_PLATFORM macos)
    if(NOT CMAKE_OSX_ARCHITECTURES MATCHES "arm64" OR NOT CMAKE_OSX_ARCHITECTURES MATCHES "x86_64")
        message(FATAL_ERROR "The macOS manifest requires a universal binary. Configure with -DCMAKE_OSX_ARCHITECTURES=arm64;x86_64")
    endif()
    set(WEBFRAME_ARCH universal)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(WEBKIT REQUIRED IMPORTED_TARGET webkit2gtk-4.1)
    pkg_check_modules(X11 REQUIRED IMPORTED_TARGET x11)
    target_sources(webframe PRIVATE src/linux.cpp)
    target_link_libraries(webframe PRIVATE PkgConfig::WEBKIT PkgConfig::X11)
    set(WEBFRAME_PLATFORM linux)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
        set(WEBFRAME_ARCH arm64)
    else()
        set(WEBFRAME_ARCH x86_64)
    endif()
endif()
set_target_properties(webframe PROPERTIES PREFIX "" OUTPUT_NAME "webframe.${WEBFRAME_PLATFORM}.${WEBFRAME_ARCH}")
set(WEBFRAME_OUTPUT "${CMAKE_CURRENT_SOURCE_DIR}/../bin")
file(MAKE_DIRECTORY "${WEBFRAME_OUTPUT}/licenses")
configure_file("${godot_cpp_SOURCE_DIR}/LICENSE.md" "${WEBFRAME_OUTPUT}/licenses/godot-cpp.txt" COPYONLY)
if(WIN32)
    file(GLOB_RECURSE WEBVIEW_LICENSES "${webview2_SOURCE_DIR}/*LICENSE*" "${webview2_SOURCE_DIR}/*License*")
    foreach(LICENSE_FILE IN LISTS WEBVIEW_LICENSES)
        get_filename_component(LICENSE_NAME "${LICENSE_FILE}" NAME)
        configure_file("${LICENSE_FILE}" "${WEBFRAME_OUTPUT}/licenses/webview2-${LICENSE_NAME}" COPYONLY)
    endforeach()
endif()
set_target_properties(webframe PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${WEBFRAME_OUTPUT}" RUNTIME_OUTPUT_DIRECTORY "${WEBFRAME_OUTPUT}")
foreach(CONFIG DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(webframe PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${CONFIG} "${WEBFRAME_OUTPUT}" RUNTIME_OUTPUT_DIRECTORY_${CONFIG} "${WEBFRAME_OUTPUT}")
endforeach()
